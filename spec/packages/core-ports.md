# core: kernel, ports, testing fakes

Part of the [`@workspace/core` spec](core.md). Covers `src/kernel/`,
`src/ports/` and `src/testing/`.

All ports are synchronous or Promise based, never import `node:*`, and use
web-standard `ReadableStream` / `WritableStream` / `AbortSignal` /
`TextEncoder`. Paths given to ports are POSIX-style strings (fakes normalize
`\` to `/`).

### result.ts

```ts
export type Ok<T> = { readonly ok: true; readonly value: T }
export type Err<E> = { readonly ok: false; readonly error: E }
export type Result<T, E = EngineError> = Ok<T> | Err<E>
ok(value) / err(error) / isOk / isErr

mapErr(result, fn) // failure only; Ok returned by identity
andThen(result, fn) // fn returns Result
unwrapOr(result, fallback)
```

### event-bus.ts

In-process domain bus between bounded contexts (not the FCP UpdateBus).

```ts
export type RunFinishedEvent = {
  runId: string
  workflowId: string
  status: "succeeded" | "failed" | "cancelled"
  cardId?: string // omitted for ad-hoc runs
}
export type DomainEventMap = { "run.finished": RunFinishedEvent }
export class EventBus {
  on<K>(type: K, handler): Disposable
  emit<K>(type: K, payload): void
}
export function createEventBus(): EventBus
```

- Handlers are stored in a `Map<type, Set<fn>>`. `dispose()` removes the
  handler and deletes the map entry when its set becomes empty.
- `emit` iterates a snapshot copy (`[...set]`) synchronously, so handlers
  added/removed during emit do not affect the current dispatch. Exceptions
  propagate to the emitter (no catching).

### hash.ts (export `./kernel/hash`)

`hashUtf8Text(text: string): string` — lowercase hex SHA-256 of the UTF-8
encoding of `text`, implemented in pure TS (no `crypto`). UTF-8 encoding is
hand-rolled from UTF-16 code units: surrogate pairs (high surrogate
`0xD800..0xDBFF` followed by any unit) become 4 bytes; lone low surrogates are
encoded as 3 bytes. Must equal Node `createHash("sha256")` on valid strings.
Test vectors: `""` →
`e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855`;
`"hello\nworld"` →
`26c60a61d01db5836ca70fefd44a6a016620413c8ef5f259a6c5612d4f79d3b8`.
This hash is the `DocumentStore` content hash used for `baseHash` locks
everywhere in core.

### DocumentStore (`document-store.ts`)

Text files under `FLOWAPP_HOME`; writes are atomic; optional optimistic
lock.

```ts
export type DocumentRead = { text: string; hash: string }
export type DocumentWriteResult =
  | { ok: true; text: string; hash: string }
  | { ok: false; kind: "conflict"; text: string; hash: string } // current disk text+hash
export type DocumentRemoveResult =
  { ok: true } | { ok: false; kind: "not_found" }
export type DocumentWriteOptions = { baseHash?: string }
export interface DocumentStore {
  read(path: string): DocumentRead | null
  write(
    path: string,
    text: string,
    options?: DocumentWriteOptions
  ): DocumentWriteResult
  list(glob: string): string[]
  remove(path: string): DocumentRemoveResult
}
```

Contract (`describeDocumentStoreContract`):

| Case                      | Behaviour                                                                                                        |
| ------------------------- | ---------------------------------------------------------------------------------------------------------------- |
| read missing              | `null`                                                                                                           |
| write+read                | `{ ok: true, text, hash: hashUtf8Text(text) }`; read returns `{text, hash}`                                      |
| stale `baseHash`          | `{ ok:false, kind:"conflict", text: diskText, hash: diskHash }`, file unchanged                                  |
| missing file + `baseHash` | missing file is treated as `""`, so `baseHash = hash("")` succeeds                                               |
| `list(glob)`              | `**` crosses `/`, `*` does not; sorted (`localeCompare`); `"**/*.md"` matches `/a/x.md` (leading slash included) |
| remove                    | `{ok:true}`; missing → `{ok:false, kind:"not_found"}`                                                            |

### EventStore (`event-store.ts`)

```ts
export type StoredEvent = { seq: number; event: EngineEvent } // EngineEvent from contracts
export type EventAppendOptions = { expectSeq: number }
export type EventAppendResult =
  | { ok: true; lastSeq: number; appended: StoredEvent[] }
  | { ok: false; kind: "conflict"; lastSeq: number }
export interface EventStore {
  append(
    runId: string,
    events: readonly EngineEvent[],
    options: EventAppendOptions
  ): EventAppendResult
  read(runId: string, fromSeq?: number): StoredEvent[] // seq >= fromSeq (default 0)
  tail(): number // global high-water; 0 when empty
}
```

Invariants: `seq` is a single **global** counter starting at 1 shared by all
runs; `expectSeq` is the last seq _of that run_ (0 when the run has no
events); mismatch → conflict with the run's current last seq; appending `[]`
with a correct `expectSeq` returns `{ok:true, lastSeq: current, appended: []}`;
a batch gets consecutive seqs; the fake throws
`event runId ${event.runId} !== ${runId}` when an event's `runId` differs.

### BlobStore (`blob-store.ts`)

```ts
export type BlobRef = { id: string; name: string; size: number; mime: string }
export type BlobPutMeta = { name?: string; mime?: string }
export interface BlobStore {
  put(
    scope: string,
    stream: ReadableStream<Uint8Array>,
    meta?: BlobPutMeta
  ): Promise<BlobRef>
  open(id: string): Promise<ReadableStream<Uint8Array> | null>
}
```

Ids are opaque and scoped; callers never build filesystem paths. Defaults:
`mime = "application/octet-stream"`, `name` non-empty (fake uses the id).
Same name in two scopes → distinct ids. Fake throws `blob scope required` on
empty scope; ids `blob_0`, `blob_1`, ...

### SecretStore (`secret-store.ts`, `secret-key.ts`)

```ts
export interface SecretStore {
  get(key): string | null
  set(key, value): void
  delete(key): void
}
export function assertSecretKey(key: string): void
```

`assertSecretKey` throws `Error("invalid secret key: <key>")` when the key
contains NUL, starts with `/` or `\`, matches `^[A-Za-z]:[\\/]`, or, split on
`/` or `\`, has an empty segment or a `..` segment. All three methods validate.
`delete` of a missing key is a no-op.

### AgentRuntime (`agent-runtime.ts`)

Implemented by `packages/agent-acp`.

```ts
export type ContentBlock =
  | { type: "text"; text: string }
  | { type: "resource"; uri: string; mimeType?: string; text?: string }
  | { type: "resource_link"; uri: string; name?: string; mimeType?: string }
export type PlanEntry = { content: string; status?: string; priority?: string }
export type TodoEntry = { id: string; content: string; status: string }
export type Money = { amount: number; currency: string }
export type SessionSpec = {
  profileId: string
  cwd: string
  mode?: string
  model?: string
  effort?: string
  mcpServers?: readonly unknown[]
  skills?: readonly Skill[] // from skills/catalog.ts
  handlePermission?: (params: unknown) => Promise<unknown>
  handleAskQuestion?: (params: unknown) => Promise<unknown>
  handleCreatePlan?: (params: unknown) => Promise<unknown>
}
export type AgentProfileInfo = {
  id: string
  modes?: readonly string[]

  version?: string | null

export type AgentEvent =
  | { kind: "message_chunk" | "thought_chunk"; text: string }

  | { kind: "tool_call_update"; id; status; content?; input?; output?: unknown }
  | { kind: "plan"; entries: PlanEntry[] }
  | { kind: "todos"; entries: TodoEntry[] }
  | { kind: "subagent"; id; description; subagentType; durationMs?: number }
  | { kind: "usage"; used: number; size: number; cost?: Money }
  | {
      kind: "stop"
      reason:
        | "end_turn"
        | "max_tokens"
        | "max_turn_requests"

    }

  readonly id: string
```

Contract context types:

```ts
type FileWatcherTestCtx = {
  watcher
  touch(path, content?): void
  ready(): Promise<void>
  waitForDebounce(): Promise<void>
}
type ProcessRunnerTestCtx = {
  runner
  sandboxRoot: string
  echoSpec(text)
  exitSpec(code, stderr?)
  hangSpec()
}
```

Process-runner contract: echo → exit 0 and stdout contains text, `pid > 0`;
`exitSpec(7,"boom")` → 7 and stderr contains `boom`; abort a hang → exit
`!== 0`; with a sandbox profile `sandboxed` is boolean (and `true` for the
fake). File-watcher contract: match, non-match ignored, one notification per
path for rapid touches, dispose, multiple globs (`/workflow.yaml` and
`/memory/x.md` notified, `/memory/x.txt` not).
